    được} $\rightarrow$ khó so sánh chi phí, chuyển đổi, doanh thu
    giữa các chiến dịch, làm giảm chất lượng quyết định
    \textbf{tối ưu chiến dịch} và \textbf{phân bổ ngân sách}.
  \end{block}
  \vspace{0.2cm}
  \begin{enumerate}
    \setlength{\itemsep}{0.4em}
    \item \textbf{Dữ liệu phân tán:} thiếu tích hợp, chuẩn hóa thủ công, dễ sai sót
    \item \textbf{Chỉ số không thống nhất:} khó đánh giá chiến dịch
      toàn diện
    \item \textbf{Thiếu góc nhìn hành vi:} phễu mua hàng, phân khúc
      khách hàng
    \item \textbf{Báo cáo chậm, thiếu nhất quán:} khó dự báo, phân
      bổ ngân sách thiếu căn cứ
  \end{enumerate}
\end{frame}

\begin{frame}{Hệ thống chỉ số KPI}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
    \begin{block}{A. Lưu lượng \& tương tác}
      \scriptsize
      \begin{itemize}
        \item \textbf{Users}: số người dùng phân biệt; \textbf{Sessions}: số phiên truy cập
        \item \textbf{Sessions/user} và số lượt xem trang (\texttt{page\_view})
        \item \textbf{Engagement rate}: tỷ lệ phiên có tương tác
        \item \textbf{New-user rate}: tỷ lệ người dùng mới (\texttt{first\_visit})
      \end{itemize}
      \vspace{0.6cm}
    \end{block}

    \column{0.32\textwidth}
    \begin{block}{B. Phễu chuyển đổi (6 bước)}
      \scriptsize
      \textbf{view\_item $\rightarrow$ cart $\rightarrow$ checkout $\rightarrow$ shipping $\rightarrow$ payment $\rightarrow$ purchase}
      \medskip
      \begin{itemize}
        \item Tỷ lệ chuyển đổi từng bước: cộng dồn tử số/mẫu số sau khi lọc
        \item \textbf{Hoàn tất mua hàng}: phiên có cả \texttt{payment} và đơn hợp lệ / phiên có \texttt{payment}
        \item \textbf{Rời bỏ giỏ hàng}: $1 -$ (phiên có cả \texttt{cart} và đơn hợp lệ / phiên có \texttt{cart})
      \end{itemize}
    \end{block}

    \column{0.32\textwidth}
    \begin{block}{C. Chuyển đổi \& doanh thu}
      \scriptsize
      \begin{itemize}
        \item \textbf{Conversion rate}: tỷ lệ phiên có giao dịch hợp lệ
        \item \textbf{AOV}: doanh thu / số đơn hàng hợp lệ
        \item Doanh thu trung bình trên mỗi phiên và mỗi người dùng
        \item \textbf{Revenue share}: tỷ trọng doanh thu theo tập dữ liệu đang lọc
      \end{itemize}
      \vspace{0.6cm}
    \end{block}
  \end{columns}

  \begin{alertblock}{Quy tắc giao dịch hợp lệ \& phạm vi chỉ số}
    \scriptsize
    \textbf{Giao dịch hợp lệ:} sự kiện \texttt{purchase} có \texttt{transaction\_id} hợp lệ, doanh thu $> 0$, và khử trùng bằng cách giữ bản ghi sớm nhất theo từng cặp (người dùng, giao dịch). \\
    \textbf{Phạm vi:} Do dữ liệu gốc không có chi phí quảng cáo và giá vốn, hệ thống không tính lợi nhuận, CPA hay ROAS.
  \end{alertblock}
\end{frame}

\begin{frame}{Chức năng chính của ứng dụng DSS}

      \begin{enumerate}

        \item \textbf{Bộ lọc:} chọn tập dữ liệu trước phân tích
