\begin{frame}{AWS IoT Core -- Kết nối thiết bị với đám mây}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
    \begin{itemize}
      \item \textbf{Định nghĩa:} dịch vụ được quản lý hoàn toàn,
        giúp \textbf{thiết bị IoT kết nối an toàn} với đám mây
        và với nhau, không cần tự dựng máy chủ.
      \item \textbf{Giao thức:} \textbf{MQTT}, HTTPS, WebSocket.
      \item \textbf{Message broker:} nhận và \textbf{định tuyến
        tin nhắn} theo mô hình publish/subscribe.
      \item \textbf{Rules Engine:} lọc dữ liệu bằng câu lệnh
        giống SQL, rồi \textbf{chuyển tới dịch vụ khác}
        (Lambda, DynamoDB, S3, SNS, \ldots).
      \item \textbf{Bảo mật:} xác thực thiết bị bằng
        \textbf{chứng chỉ X.509}, phân quyền bằng policy.
    \end{itemize}
    \column{0.38\textwidth}
    \begin{block}{Mô hình Publish / Subscribe}
      \footnotesize
      \begin{itemize}
        \item \textbf{Thiết bị} publish lên một \textit{topic}
        \item \textbf{Broker} nhận và chuyển tiếp
        \item \textbf{Bên đăng ký} (subscriber) nhận tin
      \end{itemize}
      \vspace{0.1cm}
      \textit{Ví dụ:} cảm biến publish lên topic
      \texttt{sensor/temp}, Rule lọc nhiệt độ vượt ngưỡng
      rồi gọi Lambda hoặc SNS.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Amazon SNS -- Simple Notification Service}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
    \begin{itemize}
      \item \textbf{Định nghĩa:} dịch vụ \textbf{nhắn tin
        publish/subscribe} được quản lý hoàn toàn, dùng để
        \textbf{phân phối thông báo và cảnh báo}.
      \item \textbf{Topic:} kênh nhận tin từ \textbf{publisher},
        mỗi topic có nhiều \textbf{subscriber}.
      \item \textbf{Fan-out:} một tin nhắn được \textbf{gửi đồng
        thời} tới nhiều đích.
      \item \textbf{Đích nhận:} Email, SMS, mobile push, HTTP/S,
        Lambda, SQS, \ldots
      \item \textbf{Cơ chế đẩy:} gửi ngay khi có tin,
        người nhận không cần hỏi lại.
    \end{itemize}
    \column{0.38\textwidth}
    \begin{block}{Luồng xử lý}
      \centering
      \footnotesize
      Publisher (IoT Core, Lambda)\\
      $\downarrow$ \\
      \textbf{SNS Topic}\\
      $\downarrow$ \\
      Email $\bullet$ SMS $\bullet$ Lambda $\bullet$ SQS
    \end{block}
  \end{columns}
\end{frame}

% ============ AWS IAM ============
\begin{frame}{AWS IAM -- Identity and Access Management}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
    \begin{itemize}
      \item \textbf{Định nghĩa:} dịch vụ \textbf{quản lý danh tính
        và quyền truy cập} -- quyết định \textbf{ai} được làm
        \textbf{gì} trên \textbf{tài nguyên nào} trong AWS.
      \item \textbf{User:} danh tính của một người hoặc ứng dụng,
        có thông tin đăng nhập riêng.
      \item \textbf{Role:} danh tính \textbf{tạm thời} để dịch vụ
        (ví dụ Lambda) hoặc user \textbf{đảm nhận} quyền.
      \item \textbf{Policy:} tài liệu JSON mô tả quyền
        \textbf{cho phép/từ chối} hành động.
      \item \textbf{Nguyên tắc:} cấp \textbf{quyền tối thiểu}
        (least privilege).
    \end{itemize}
    \column{0.38\textwidth}
    \begin{block}{Mối quan hệ}
      \footnotesize
      \begin{itemize}
        \item \textbf{User / Role} $\rightarrow$ \textit{ai}
        \item \textbf{Policy} $\rightarrow$ \textit{được làm gì}
        \item Gắn \textbf{Policy} vào \textbf{User/Role} để cấp quyền
      \end{itemize}
      \vspace{0.1cm}
      \textit{Ví dụ:} Role cho Lambda + Policy chỉ cho phép
      \texttt{dynamodb:PutItem}.
    \end{block}
  \end{columns}
\end{frame}

% ============ AWS LAMBDA ============
\begin{frame}{AWS Lambda -- Function as a Service}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
    \begin{itemize}
      \item \textbf{Định nghĩa:} dịch vụ \textbf{chạy code không
        cần quản lý máy chủ} (serverless compute).
      \item \textbf{Chức năng:} tự động chạy function khi có
        \textbf{sự kiện kích hoạt} (trigger) từ API Gateway, S3,
        SQS, EventBridge, \ldots
      \item \textbf{Tự động mở rộng:} từ vài đến hàng nghìn
        request đồng thời.
      \item \textbf{Trả phí theo mức dùng:} tính theo số lần gọi
        và thời gian chạy (ms).
      \item \textbf{Hỗ trợ:} Python, Node.js, Java, Go, \ldots
    \end{itemize}
    \column{0.38\textwidth}
    \begin{block}{Cách hoạt động}
      \footnotesize
      \begin{enumerate}
        \item \textbf{Sự kiện} xảy ra (request, file mới, lịch)
        \item AWS \textbf{khởi chạy} function
        \item Function \textbf{xử lý} và trả kết quả
        \item Chạy xong thì \textbf{dừng}, không tính phí chờ
      \end{enumerate}
    \end{block}
  \end{columns}
\end{frame}

% ============ API GATEWAY ============
\begin{frame}{Amazon API Gateway}
  \small
  \begin{columns}[onlytextwidth]
    \column{0.58\textwidth}
    \begin{itemize}
      \item \textbf{Định nghĩa:} dịch vụ \textbf{tạo, công bố và
        quản lý API} (REST, HTTP, WebSocket) ở quy mô lớn.
      \item \textbf{Chức năng chính:} làm \textbf{cổng vào} --
        nhận request từ client và \textbf{chuyển tới Lambda}
        hoặc backend khác.
      \item \textbf{Tính năng đi kèm:} xác thực, giới hạn lưu
        lượng (throttling), cache, giám sát.
      \item \textbf{Vai trò trong serverless:} kết nối
        \textbf{Internet $\leftrightarrow$ Lambda}.
    \end{itemize}
    \column{0.38\textwidth}
    \begin{block}{Luồng xử lý}
      \footnotesize
      \centering
      Client\\
      $\downarrow$ \\ \texttt{GET /items}\\
      \textbf{API Gateway}\\
      $\downarrow$ \\ kiểm tra, giới hạn lưu lượng\\
      \textbf{Lambda}\\
      $\downarrow$ \\
      Phản hồi JSON về Client
    \end{block}
  \end{columns}
\end{frame}
